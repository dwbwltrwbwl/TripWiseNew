GO

GO

/****** Создаем представление для удобного просмотра активных токенов ******/
CREATE VIEW [dbo].[vActiveAuthTokens]
AS
SELECT 
    ut.Id,
    ut.UserId,
    u.email AS UserEmail,
    CONCAT(u.last_name, ' ', u.first_name) AS UserFullName,
    ut.Token,
    ut.CreatedAt,
    ut.ExpiresAt,
    DATEDIFF(DAY, GETUTCDATE(), ut.ExpiresAt) AS DaysUntilExpiration
FROM [dbo].[UserAuthTokens] ut
INNER JOIN [dbo].[Users] u ON ut.UserId = u.idUser
WHERE ut.ExpiresAt > GETUTCDATE()
GO
